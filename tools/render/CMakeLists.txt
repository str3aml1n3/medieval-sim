cmake_minimum_required(VERSION 3.10)
project(render_music C)
set(DM_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(DM_ENABLE_ASAN OFF CACHE BOOL "" FORCE)
set(DM_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(../dmusic dmusic)
add_executable(render-music render-music.c)
target_link_libraries(render-music PRIVATE dmusic)
if (MINGW)
  target_link_options(render-music PRIVATE -static)
endif()
